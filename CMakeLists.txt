cmake_minimum_required(VERSION 3.20)
option(BULB_HOST_TESTS "Build portable application policy/codec tests" OFF)
set(MATTER_BULB_STACK_DIR "${CMAKE_CURRENT_LIST_DIR}/matter" CACHE PATH "Pinned Matter checkout")
if(BULB_HOST_TESTS)
  project(MatterBulbHost LANGUAGES C)
  enable_testing()
  foreach(test factory_codec)
    add_executable(test_${test} main/${test}.c tests/test_${test}.c)
    target_include_directories(test_${test} PRIVATE main)
    target_compile_features(test_${test} PRIVATE c_std_99)
    target_compile_options(test_${test} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME ${test} COMMAND test_${test})
  endforeach()
  add_executable(qr_grid tools/qr_grid.c "${MATTER_BULB_STACK_DIR}/third_party/qrcodegen/qrcodegen.c")
  target_include_directories(qr_grid PRIVATE "${MATTER_BULB_STACK_DIR}/third_party/qrcodegen")
  add_executable(test_bulb_store main/bulb_store.c main/factory_codec.c tests/test_bulb_store.c)
  target_include_directories(test_bulb_store PRIVATE tests/idf-stub main "${MATTER_BULB_STACK_DIR}/main/apps/matter" "${MATTER_BULB_STACK_DIR}/components/matter_c_esp32")
  target_compile_options(test_bulb_store PRIVATE -Wall -Wextra -Werror -UNDEBUG "-include${CMAKE_CURRENT_LIST_DIR}/tests/test_store_alloc.h")
  add_test(NAME bulb_store COMMAND test_bulb_store)
  add_executable(test_bulb_clock main/bulb_clock.c tests/test_bulb_clock.c "${MATTER_BULB_STACK_DIR}/main/apps/matter/codec/matter_tlv.c")
  target_include_directories(test_bulb_clock PRIVATE main "${MATTER_BULB_STACK_DIR}/main/apps/matter")
  target_compile_options(test_bulb_clock PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME bulb_clock COMMAND test_bulb_clock)
  add_executable(test_bulb_country main/bulb_country.c tests/test_bulb_country.c)
  target_include_directories(test_bulb_country PRIVATE main)
  target_compile_options(test_bulb_country PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME bulb_country COMMAND test_bulb_country)
  return()
endif()
set(EXTRA_COMPONENT_DIRS "${MATTER_BULB_STACK_DIR}/components/matter_c" "${MATTER_BULB_STACK_DIR}/components/matter_c_esp32")
if(BULB_MDNS_DIR)
  list(APPEND EXTRA_COMPONENT_DIRS "${BULB_MDNS_DIR}")
endif()
set(PROJECT_VER "1" CACHE STRING "Monotonically increasing numeric Matter SoftwareVersion")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(matter_c_bulb)
